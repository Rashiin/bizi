var total = 0;
var items = Me.getXPath("PurchaseRequest.Items");
for (var i = 0; i < items.size(); i++)
{
    var item = items.get(i);
    var line = CHelper.ToDecimal(item.getXPath("Quantity")) * CHelper.ToDecimal(item.getXPath("UnitPrice"));
    item.setXPath("LineTotal", line);
    total = total + line;
}
<PurchaseRequest.TotalAmount> = total;
